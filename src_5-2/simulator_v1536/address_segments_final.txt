# 最终配置：段序号即变量名（1~12），不要seg前缀
# 0区1区存1，3区4区存字母表每个都从A开始
# 格式：段序号 分区 起始地址 长度 [名称] [数据类型] [初始值]
# 长度：0区1区是位数，3区4区是寄存器数量

# 0区（线圈）- 存1
1  0  0    1    1    bool  1
2  0  8    1    2    bool  1
3  0  16   1    3    bool  1
4  0  24   1    4    bool  1

# 1区（离散输入）- 存1
5  1  0    1    5    bool  1
6  1  8    1    6    bool  1
7  1  16   1    7    bool  1
8  1  32   1    8    bool  1

# 3区（输入寄存器）- 存字母表从A开始
# 字符串"ABCDEFGHIJKLMN"是14字符，需要7个寄存器(14字节)
9  3  0    7    9    string  ABCDEFGHIJKLMN
10 3  13   7    10   string  ABCDEFGHIJKLMN

# 4区（保持寄存器）- 存字母表从A开始
# 字符串"ABCDEFGHIJKLMN"是14字符，需要7个寄存器(14字节)
11 4  0    7    11   string  ABCDEFGHIJKLMN
12 4  13   7    12   string  ABCDEFGHIJKLMN
